mod platform {
    pub(super) async fn signal() {
        tokio::select! {
            _ = super::ctrl_c() => {},
            _ = terminate() => {},
        }
    }

    async fn terminate() {
        let mut signal =
            match tokio::signal::unix::signal(tokio::signal::unix::SignalKind::terminate()) {
                Ok(signal) => signal,
                Err(error) => {
                    tracing::warn!(error = %error, "SIGTERM shutdown signal unavailable...");

                    return;
                }
            };
        signal.recv().await;
    }
}

### 1.4 AtomCode — 信号处理与终端守护

**核心文件**：`crates/atomcode-tuix/src/signal_restore.rs`

```rust
// crates/atomcode-tuix/src/signal_restore.rs (L58-85)
pub(crate) fn arm() {
    if INSTALLED.swap(true, Ordering::SeqCst) {
        return;
    }
    unsafe {
        if libc::tcgetattr(
            libc::STDIN_FILENO,
            addr_of_mut!(ORIG_TERMIOS).cast::<libc::termios>(),
        ) == 0 {
            TERMIOS_SAVED.store(true, Ordering::SeqCst);
        }
        let mut sa: libc::sigaction = core::mem::zeroed();
        sa.sa_sigaction = handler as *const () as libc::sighandler_t;
        libc::sigemptyset(&mut sa.sa_mask);
        libc::sigaddset(&mut sa.sa_mask, libc::SIGTERM);
        libc::sigaddset(&mut sa.sa_mask, libc::SIGINT);
        libc::sigaddset(&mut sa.sa_mask, libc::SIGHUP);
        sa.sa_flags = 0;
        for sig in [libc::SIGTERM, libc::SIGINT, libc::SIGHUP] {
            libc::sigaction(sig, &sa, core::ptr::null_mut());
        }
    }
}
```

**关键设计**：
- 使用 `sigaction` 而非 tokio signal（避免 wedge 时信号被饿死）
- 仅使用 async-signal-safe 调用（`write`、`tcsetattr`、`raise`）
- 信号处理器内恢复终端 → 重置 termios → 重新 raise 信号
- 屏蔽 SIGTERM/SIGINT/SIGHUP 防止重入

  const clientTtlOption = this[kOptions].clientTtl;

  if (this[kClients].length === 0) {
    const dispatcher = this[kFactory](this[kUrl], this[kOptions]);
    this[kAddClient](dispatcher);
    return dispatcher;
  }

  let checked = 0;
  while (checked < this[kClients].length) {
    this[kIndex] = (this[kIndex] + 1) % this[kClients].length;
    const client = this[kClients][this[kIndex]];

    if (clientTtlOption != null && clientTtlOption > 0 && client.ttl && ((Date.now() - client.ttl) > clientTtlOption)) {

      this[kIndex]--;
      continue;

    if (!client[kNeedDrain]) {
      return client;
    }
    checked++;
  }

  if (!this[kConnections] || this[kClients].length < this[kConnections]) {

private handleClose(closeCode: number): void {
  this.stopPingInterval();
  if (this.state === 'closed') return;
  this.ws = null;
  const previousState = this.state;
  this.state = 'closed';

  if (PERMANENT_CLOSE_CODES.has(closeCode)) {
    this.callbacks.onClose?.();
    return;
  }

  if (closeCode === 4001) {
    this.sessionNotFoundRetries++;
    if (this.sessionNotFoundRetries > MAX_SESSION_NOT_FOUND_RETRIES) {
      this.callbacks.onClose?.();
      return;
    }
    this.scheduleReconnect(RECONNECT_DELAY_MS * this.sessionNotFoundRetries, `4001 attempt ${this.sessionNotFoundRetries}/${MAX_SESSION_NOT_FOUND_RETRIES}`);
    return;
  }

  if (previousState === 'connected' && this.reconnectAttempts < MAX_RECONNECT_ATTEMPTS) {
    this.reconnectAttempts++;
    this.scheduleReconnect(RECONNECT_DELAY_MS, `attempt ${this.reconnectAttempts}/${MAX_RECONNECT_ATTEMPTS}`);
  }
}

pub(crate) fn frame_stream(
    stream: RawEventStream,
    target_format: WireFormat,
) -> Sse<SseFrameStream> {
    let framed = async_stream::stream! {
        let mut stream = stream;
        let mut failed = false;
        while let Some(item) = futures_util::StreamExt::next(&mut stream).await {
            let event = match item {
                Ok(value) => match frame_event(target_format, value) {
                    Ok(event) => event,
                    Err(error) => {
                        failed = true;
                        error_event(target_format, error.to_string())
                    }
                },
                Err(LlmStreamError::Upstream(value)) => {
                    failed = true;
                    frame_event(target_format, value.clone()).unwrap_or_else(|error| {
                        error_event(target_format, value.to_string())
                    })
                }
                Err(LlmStreamError::Client(error)) => {
                    failed = true;
                    error_event(target_format, error.to_string())
                }
            };
            yield Ok(event);
            if failed { break; }
        }

        if !failed && target_format == WireFormat::OpenAiChat {
            yield Ok(Event::default().data("[DONE]"));
        }
    };
    Sse::new(Box::pin(framed) as SseFrameStream)
}

## 8. 与 laew 对比

### 8.1 laew 当前架构

laew 是**单进程 CLI**，特点：
- 无 Daemon/Server 模式
- 无服务发现
- 无多实例部署
- 无 API 网关
- 无 WebSocket/SSE 服务
- 无容器化部署
- 配置存储在 SQLite（`LsmAgentEmergentWork.db`）
- 无动态配置推送

### 8.2 各工程分布式能力对比矩阵

| 维度 | atomcode | claudecode | deepseek | openclaw | opencode | pi | undici | Switchyard | laew |
|------|----------|------------|----------|----------|----------|-----|--------|------------|------|
| Daemon/Server | ❌ (TUI only) | ❌ (远程客户端) | ⚠️ (WebWorker) | ✅ (Gateway) | ✅ (Server) | ⚠️ (RPC) | ❌ (客户端) | ✅ (Server) | ❌ |
| 服务发现 | ❌ | ❌ | ❌ | ✅ (Tailscale) | ✅ (Workspace) | ✅ (Radius) | ✅ (DNS/Proxy) | ❌ (静态) | ❌ |
| 多实例 | ✅ (Team) | ❌ | ✅ (Worker) | ✅ (Gateway) | ✅ (Instance) | ❌ | ✅ (Pool) | ✅ (Route) | ❌ |
| API 网关 | ❌ | ❌ | ❌ | ✅ (多层) | ✅ (HTTP API) | ❌ | ❌ | ✅ (完整) | ❌ |
| WebSocket/SSE | ❌ | ✅ (WS) | ✅ (Tunnel) | ✅ (WS) | ✅ (SSE) | ❌ | ❌ | ✅ (SSE) | ❌ |
| 容器化 | ❌ | ❌ | ✅ (Worker) | ✅ (Tailscale) | ✅ (SST/CF) | ❌ | ❌ | ✅ (TLS) | ❌ |
| 配置中心 | ⚠️ (文件) | ❌ | ⚠️ (镜像) | ✅ (动态) | ✅ (Env+DB) | ⚠️ (文件) | ❌ | ✅ (TOML) | ⚠️ (SQLite) |

### 8.3 laew 可扩展的分布式架构路径

基于 8 个工程的分析，laew 向分布式演进的可行路径：

1. **Daemon 模式**：参考 Switchyard 的 `tokio::signal` + `tokio::select!` 实现优雅关闭
2. **服务发现**：参考 Undici 的 `EnvHttpProxyAgent` 实现代理链 + DNS 解析

4. **API 网关**：参考 Switchyard 的 7 种路由算法 + OpenClaw 的安全策略
5. **WebSocket/SSE**：参考 ClaudeCode 的重连策略 + Switchyard 的 SSE 帧流
6. **容器化**：参考 OpenCode 的 SST + Cloudflare 部署
7. **配置中心**：参考 OpenClaw 的动态配置重载 + Switchyard 的 TOML 部署配置
