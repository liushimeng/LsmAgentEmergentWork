// loadSkillsDir.ts - Dynamic skill discovery
export async function loadDynamicSkills(): Promise<void> {
  const dirs = getSkillDirs()  // Collect all settings-level skills directories
  const loadedSkills = await Promise.all(
    dirs.map(dir => loadSkillsFromSkillsDir(dir, 'projectSettings'))

  // Reverse processing: deeper paths override shallower ones
  for (let i = loadedSkills.length - 1; i >= 0; i--) {
    for (const { skill } of loadedSkills[i] ?? []) {
      if (skill.type === 'prompt') {
        dynamicSkills.set(skill.name, skill)
      }
    }
  }
  skillsLoaded.emit()  // Notify listeners to clear cache
}

## 12. undici - Interceptor Chain + WASM Dynamic Loading

### 12.1 Interceptor Chain Architecture

undici's interceptor chain implements a lightweight AOP extension mechanism with **onion model + function composition**.

#### Core compose Implementation

```javascript
// dispatcher/dispatcher.js - Interceptor chain composition
class Dispatcher extends EventEmitter {
  compose(...args) {
    const interceptors = Array.isArray(args[0]) ? args[0] : args
    let dispatch = this.dispatch.bind(this)

    for (const interceptor of interceptors) {
      if (interceptor == null) continue
      if (typeof interceptor !== 'function')
        throw new TypeError(`invalid interceptor, expected function received ${typeof interceptor}`)
      dispatch = interceptor(dispatch)  // Function composition
      if (dispatch == null || typeof dispatch !== 'function' || dispatch.length !== 2)
        throw new TypeError('invalid interceptor')
    }

    // Proxy wrapper: inject origin
    return new Proxy(this, {
      get: (target, key) => key === 'dispatch' ? dispatch : target[key]
    })
  }
}
```

    ? require('../llhttp/llhttp-wasm.js')
