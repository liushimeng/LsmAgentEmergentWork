function makeCacheKey (opts) {
  const origin = opts.origin ? opts.origin.toString() : ''
  let fullPath = opts.path || '/'
  if (opts.query && !pathHasQueryOrFragment(fullPath)) {

#### 请求去重 (`lib/interceptor/deduplicate.js`)

```javascript
function makeDeduplicationKey (cacheKey, excludeHeaders) {
  const headers = {}
  if (cacheKey.headers) {
    const sortedHeaders = Object.keys(cacheKey.headers).sort()
    for (const header of sortedHeaders) {
      if (excludeHeaders?.has(header.toLowerCase())) continue
      headers[header] = cacheKey.headers[header]
    }
  }
  return JSON.stringify([cacheKey.origin, cacheKey.method, cacheKey.path, headers])
}
```

**设计要点**：
- 排序 header 键确保一致性
- 支持排除特定 header（如 `Authorization`）
- JSON.stringify 避免分隔符冲突（修复 issue #5012）

### 1.3 Switchyard — 前缀指纹缓存

#### 前缀探测 (`crates/switchyard-server/src/stats/cache_eligibility.rs`)

```rust
pub(crate) struct PrefixProbe {
    boundaries: Vec<(u64, u64)>,  // (byte_len, hash)
    total_len: u64,
}

impl PrefixProbe {
    pub(crate) fn eligible_fraction(&self, seen: &HashSet<u64>) -> f64 {
        let eligible = self.boundaries.iter()
            .filter(|(_, hash)| seen.contains(hash))
            .map(|(len, _)| *len)
            .max()
            .unwrap_or(0);
        eligible as f64 / self.total_len as f64
    }
}

pub(crate) fn prefix_probe(body: &Value) -> PrefixProbe {
    // 1. hash system + instructions → 第一个 boundary
    // 2. 逐条 hash messages/input → 递增 boundary
    // 每个 boundary 累积 byte_len + running hash
}
```

**设计要点**：
- 环境变量 `SWITCHYARD_THEORETICAL_CACHE=1` 启用
- 格式无关（同时处理 `messages` 和 `input` 字段）
- 增量哈希：每个 boundary 的 hash 包含之前所有内容
- `eligible_fraction` = 已见前缀 / 总长度 = 理论 KV-cache 命中率

### 2.1 claudecode — LRU + 字节大小限制

#### FileStateCache (`src/utils/fileStateCache.ts`)

```typescript
import { LRUCache } from 'lru-cache'

export const READ_FILE_STATE_CACHE_SIZE = 100
const DEFAULT_MAX_CACHE_SIZE_BYTES = 25 * 1024 * 1024  // 25MB

export class FileStateCache {
  private cache: LRUCache<string, FileState>

  constructor(maxEntries: number, maxSizeBytes: number) {
    this.cache = new LRUCache<string, FileState>({
      max: maxEntries,
      maxSize: maxSizeBytes,
      sizeCalculation: value => Math.max(1, Buffer.byteLength(value.content)),
    })
  }

  get(key: string): FileState | undefined {
    return this.cache.get(normalize(key))
  }
}
```

**设计要点**：
- 基于 `lru-cache` 库，支持 max entries + max bytes 双限制
- 路径归一化（`normalize(key)`）确保相对/绝对路径命中
- 大小计算基于 `Buffer.byteLength(value.content)`
- 支持 `dump()`/`load()` 序列化（用于 SubAgent 上下文传递）
- 支持 `mergeFileStateCaches`（按 timestamp 合并）

#### FileReadCache (`src/utils/fileReadCache.ts`)

```typescript
class FileReadCache {
  private cache = new Map<string, CachedFileData>()
  private readonly maxCacheSize = 1000

  readFile(filePath: string): { content: string; encoding: BufferEncoding } {
    const stats = fs.statSync(filePath)
    const cachedData = this.cache.get(cacheKey)
    if (cachedData && cachedData.mtime === stats.mtimeMs) {
      return { content: cachedData.content, encoding: cachedData.encoding }
    }
    // Cache miss → read file → update cache
    if (this.cache.size > this.maxCacheSize) {
      const firstKey = this.cache.keys().next().value
      this.cache.delete(firstKey)  // FIFO 驱逐
    }
  }
}
```

**设计要点**：
- 基于 `Map` 的 FIFO 驱逐（非 LRU）
- mtime 失效：`cachedData.mtime === stats.mtimeMs`
- 文件删除时自动 `delete(filePath)` + re-throw
- 最大 1000 条目

### 2.2 atomcode — Mutex&lt;HashMap&gt; + Instant TTL

#### PasswordCache (`crates/atomcode-capabilities/src/askpass/cache.rs`)

```rust
use std::collections::HashMap;
use std::sync::Mutex;
use std::time::{Duration, Instant};
use zeroize::Zeroizing;

pub struct PasswordCache {
    ttl: Duration,
    inner: Mutex<HashMap<String, (Zeroizing<String>, Instant)>>,
}

impl PasswordCache {
    pub fn get(&self, key: &str, now: Instant) -> Option<Zeroizing<String>> {
        let mut m = self.inner.lock().unwrap();
        match m.get(key) {
            Some((pw, at)) if now.duration_since(*at) <= self.ttl => Some(pw.clone()),
            Some(_) => { m.remove(key); None }  // expired → evict
            None => None,
        }
    }
}
```

**设计要点**：
- `Mutex<HashMap>` — 简单有效的并发控制
- `Instant` TTL — 读时检查过期
- `Zeroizing<String>` — 安全擦除敏感数据
- 读时驱逐（lazy eviction）

private installWritePath(): void {
  this.ctx.on('session/event', (session, event) => {
    if (event.type === 'turn/end') {

## 6. Prompt/Token 缓存

### 6.1 pi — Cache Waste 检测

#### Cache Stats (`packages/coding-agent/src/core/cache-stats.ts`)

```typescript
export const CACHE_TTL_MS = 5 * 60 * 1000  // Anthropic 默认 5 分钟
const NOISE_FLOOR_TOKENS = 1024  // 低于此值的 miss 视为噪声

export interface CacheMiss {
  missedTokens: number
  missedCost: number
  idleMs: number
  modelChanged: boolean
}

function detectMiss(prev, message, models): CacheMiss | undefined {
  const promptTokens = usage.input + usage.cacheRead + usage.cacheWrite
  const missedTokens = Math.min(prev.promptTokens, promptTokens) - usage.cacheRead
  if (missedTokens <= NOISE_FLOOR_TOKENS) return undefined

  const paidPerToken = (usage.cost.input + usage.cost.cacheWrite) / (usage.input + usage.cacheWrite)
  const readPerToken = usage.cost.cacheRead / usage.cacheRead
  return {
    missedTokens,
    missedCost: missedTokens * Math.max(0, paidPerToken - readPerToken),
    idleMs: message.timestamp - prev.timestamp,
    modelChanged: `${message.provider}/${message.model}` !== prev.modelKey,
  }
}
```

**设计要点**：
- 逐 turn 检测 cache miss
- 区分真实 miss vs 噪声（`NOISE_FLOOR_TOKENS`）
- 计算额外成本（`missedCost`）
- 区分 compaction 导致的合法 miss vs 模型切换导致的真实 miss
- Sticky `reportedCache` 标记 — 区分 cache-read-only provider 和不报告 cache 的 provider
