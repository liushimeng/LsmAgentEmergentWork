```rust

fn execution_policy_from_text(text: &str) -> ExecutionPolicy {
    let lower = without_quoted_examples(&text.to_lowercase());
    let mut flags = 0;
    if contains_any(&lower, &["不要运行任何命令", "do not run any command", ...]) {
        flags |= NO_SHELL;
    }
    if contains_any(&lower, &["不要编译", "don't build", ...]) {
        flags |= NO_BUILD;
    }
    ExecutionPolicy(flags)
}
```

这是一个**运行时动态策略调整**的典型实现：用户在对话中说"不要编译"，后续所有 bash 工具调用都会被 middleware 拦截。

```typescript
// src/utils/streamJsonStdoutGuard.ts 关键代码
export function installStreamJsonStdoutGuard(): void {
  originalWrite = process.stdout.write.bind(process.stdout);
  process.stdout.write = function (chunk: string | Uint8Array, ...): boolean {
    buffer += text;
    while ((newlineIdx = buffer.indexOf('\n')) !== -1) {
      const line = buffer.slice(0, newlineIdx);
      if (isJsonLine(line)) {
        wrote = originalWrite!(line + '\n');
      } else {
        process.stderr.write(`${STDOUT_GUARD_MARKER} ${line}\n`);
      }
    }
    return wrote;
  };
}
```

**设计要点**：
- 包装 `process.stdout.write` 在 NDJSON 流中实时检测非 JSON 行
- 非 JSON 行被重定向到 stderr，不破坏 JSON 流
- 安装两次是幂等的

```typescript
export function buildContextEngineRuntimeSettings(params: {
  mode?: ContextEngineRuntimeMode;
  requestedModel?: OptionalString;
  resolvedModel?: OptionalString;
  fallbackReason?: OptionalString;
  degradedReason?: OptionalString;
  promptTokenBudget?: number | null;
}): ContextEngineRuntimeSettings {
  const resolvedViaFallback = requestedModel !== resolvedModel;
  const mode = params.mode ?? (degradedReason ? "degraded" : fallbackReason || resolvedViaFallback ? "fallback" : "normal");
  return {
    schemaVersion: 1,
    runtime: { host: "openclaw", mode, harnessId, runtimeId },
    model: { requested, resolved, provider, family },
    limits: { promptTokenBudget, maxOutputTokens },
    diagnostics: { fallbackReason, degradedReason },
  };
}
```
