### 1.2 数据流

```
用户输入
  │
  ▼
TuiSession.handle_input() / run_one_shot()
  │
  ▼
MultiAgentOrchestrator::handle()
  │ ① inject project doc
  │ ② call Yolo
  ▼
YoloRunner::run_yolo() → classification
  │
  ▼ (task_level)
  ├── simple ─→ Orchestrator::run_simple
  │              ├── SubAgentRunner::run_unit
  │              └── QualityCheck::check_subagent
  │                       │
  │                       ▼ pass
  │              SessionContext::summarize
  │
  ├── medium ─→ Orchestrator::run_medium
  │              ├── MainWorkRunner::plan_workflows
  │              ├── for each workflow:
  │              │     SubAgentRunner::run_unit
  │              │     QualityCheck::check_subagent
  │              └── SessionContext::summarize
  │
  └── hard ─→ Orchestrator::run_hard
                ├── PlanRunner::generate (write plans/*.md)
                ├── QualityCheck::check_plan
                ├── MainWorkRunner::parse_plan
                ├── for each workflow:
                │     SubAgentRunner::run_unit
                │     QualityCheck::check_subagent
                └── SessionContext::summarize

任意环节失败:
  → QualityFail { source, reason, retryable }
  → Orchestrator::retry_or_escalate
       retryable=true  → 重跑当前单元
       retryable=false → 升级到上一层
       顶层(Yolo)失败 → 输出 user_suggestion
```

## 5. Orchestrator 主循环伪代码

```rust
impl MultiAgentOrchestrator {
    pub async fn handle(&self, req: OrchestrationRequest<'_>) -> Result<OrchestrationOutcome> {
        // 0) 项目上下文首次注入
        inject_once(req.session, work_dir);

        // 1) Yolo 入口
        let classification = self.run_yolo_classification(req.session).await?;
        write_session_memory(yolo, &classification, ...);

        let mut retry_count = 0;
        loop {
            retry_count += 1;
            if retry_count > MAX_RETRY {  // 3
                let suggestion = if classification.user_suggestion_if_fail.is_empty() {
                    "请补充任务信息或调整目标".into()
                } else {
                    classification.user_suggestion_if_fail.clone()
                };
                return Ok(OrchestrationOutcome::Failed { classification, suggestion, usage });
            }

            // 2) 调度执行
            let result = match classification.task_level {
                TaskLevel::Simple => self.run_simple(&classification, req.session).await,
                TaskLevel::Medium => self.run_medium(&classification, req.session).await,
                TaskLevel::Hard   => self.run_hard(&classification, req.session).await,
            };

            match result {
                Ok(task_result) => {
                    // 3) SessionContext 收口
                    let summary = self.run_session_context(&classification, &task_result, req.session).await?;
                    let mut out = task_result;
                    out.summary = summary.text;
                    return Ok(OrchestrationOutcome::Executed { result: out, total_usage: total });
                }
                Err(QualityFail { source, reason, retryable, suggestion }) => {
                    write_session_memory(quality, &reason, ...);
                    if !retryable {
                        // 升级到上一层
                        let new_c = self.run_yolo_with_failure(&classification, &reason, req.session).await?;
                        write_session_memory(yolo, &new_c, ...);
                        // 如果 Yolo 给出 user_suggestion 且 agent_role 改变为更低档,继续循环
                        classification = new_c;
                        continue;
                    }
                    // retryable=true 留在当前档位,继续重跑
                    continue;
                }
            }
        }
    }
}
```

### 5.1 简单档

```rust
async fn run_simple(&self, c: &Classification, session: &mut Session) -> Result<TaskResult, QualityFail> {
    let sub_outcome = self.sub_agent.run_unit(&SubFlowInput {
        description: &c.goal_summary,
        expected_output: &c.decomposition_plan.join("; "),
        depends_on: vec![],
    }, session).await?;

    let qc_report = self.quality.check_subagent(c, &sub_outcome).await?;
    write_session_memory(quality, &qc_report, ...);

    if qc_report.verdict == Verdict::Pass {
        Ok(TaskResult {
            goal: c.goal_summary.clone(),
            plan_doc: None,
            workflows: vec![WorkflowResult {
                id: "wf-1".into(),
                name: "单步执行".into(),
                subflow_outcome: sub_outcome.text,
                quality_report: qc_report,
                usage: sub_outcome.usage,
            }],
            summary: String::new(),
        })
    } else {
        Err(QualityFail {
            source: AgentRole::SubAgent,
            reason: qc_report.issues.join("; "),
            retryable: qc_report.retryable,
            suggestion: qc_report.suggestion,
        })
    }
}
```

### 5.2 中等档

```rust
async fn run_medium(&self, c: &Classification, session: &mut Session) -> Result<TaskResult, QualityFail> {
    // 1) Main-Work 拆 WorkFlow
    let workflow_spec = self.main_work.plan_workflows(c, session).await?;
    write_session_memory(main, &workflow_spec, ...);

    let qc_main = self.quality.check_main(c, &workflow_spec).await?;
    write_session_memory(quality, &qc_main, ...);
    if qc_main.verdict == Fail {
        return Err(QualityFail {
            source: AgentRole::MainWork,
            reason: qc_main.issues.join("; "),
            retryable: qc_main.retryable,
            suggestion: qc_main.suggestion,
        });
    }

    // 2) 拓扑排序
    let ordered = topo_sort(&workflow_spec.workflows)?;

    // 3) 顺序执行
    let mut results = vec![];
    let mut dep_outputs: HashMap<String, String> = HashMap::new();
    for wf in ordered {
        let sub_outcome = self.sub_agent.run_unit(&SubFlowInput {
            description: &format!("{}\n\n步骤:\n{}", wf.name, wf.steps.join("\n")),
            expected_output: wf.acceptance.join("; "),
            depends_on: dep_outputs.iter()
                .filter(|(k, _)| wf.depends_on.contains(k))
                .map(|(_, v)| v.clone())
                .collect(),
        }, session).await?;

        let qc = self.quality.check_subagent_for_workflow(&wf, &sub_outcome).await?;
        write_session_memory(quality, &qc, ...);

        if qc.verdict == Fail {
            return Err(QualityFail {
                source: AgentRole::SubAgent,
                reason: format!("wf={}: {}", wf.id, qc.issues.join("; ")),
                retryable: qc.retryable,
                suggestion: qc.suggestion,
            });
        }

        dep_outputs.insert(wf.id.clone(), sub_outcome.text.clone());
        results.push(WorkflowResult {
            id: wf.id.clone(),
            name: wf.name.clone(),
            subflow_outcome: sub_outcome.text,
            quality_report: qc,
            usage: sub_outcome.usage,
        });
    }

    Ok(TaskResult {
        goal: c.goal_summary.clone(),
        plan_doc: None,
        workflows: results,
        summary: String::new(),
    })
}
```

## 6. 失败回流与 Yolo 重新评估

```rust
async fn run_yolo_with_failure(
    &self,
    prev: &TaskClassification,
    failure: &QualityFail,
    session: &mut Session,
) -> Result<TaskClassification, AgentError> {
    // 构造特殊提示
    let mut yolo_session = Session::new();
    for msg in session.context() {
        yolo_session.context_mut().push(msg.clone());
    }
    let failure_msg = format!(
        "[PREVIOUS_FAILURE]\n源: {}\n任务级别: {}\n原目标: {}\n失败原因: {}\n建议: {}\n请重新评估:可重试 → 修订 decomposition_plan 重发;不可重试 → 填 user_suggestion_if_fail 并给出 direct_answer 告知用户。",
        failure.source.as_str(),
        prev.task_level.as_str(),
        prev.goal_summary,
        failure.reason,
        failure.suggestion,
    );
    yolo_session.context_mut().push(ChatMessage::user(failure_msg));

    let (text, usage) = self.yolo.run_session(&mut yolo_session).await?;
    parse_classification(&text)
}
```

## 7. SubAgent-Work 单元执行

```rust
impl SubAgentRunner {
    pub async fn run_unit(
        &self,
        input: &SubFlowInput,
        session: &mut Session,
    ) -> Result<SubFlowOutcome, AgentError> {
        // 1) 构造独立 AgentContext
        let mut agent_ctx = AgentContext::new(AgentRole::SubAgent, session.id());
        agent_ctx.push_user(&input.to_user_prompt());

        // 2) run_session
        let mut sub_session = Session::new();
        sub_session.context_mut().push(ChatMessage::user(&input.to_user_prompt()));
        let (text, usage) = self.agent.run_session(&mut sub_session).await?;

        // 3) 写 Agent-Memory
        self.db.insert_agent_memory(&AgentMemoryEntry {
            agent_role: AgentRole::SubAgent,
            session_id: session.id().to_string(),
            input_summary: input.description.chars().take(160).collect(),
            output_summary: text.chars().take(160).collect(),
            error_summary: None,
            artifacts: json!({ "subflow_id": input.id }),
        })?;

        Ok(SubFlowOutcome { text, usage, sub_session })
    }
}
```

```rust
impl PlanRunner {
    pub async fn generate(
        &self,
        c: &TaskClassification,
        session: &mut Session,
    ) -> Result<PathBuf, AgentError> {
        // 1) 确保 plans/ 存在
        let plans_dir = self.root_dir.join("plans");
        std::fs::create_dir_all(&plans_dir)?;

        // 2) 调用 Plan Agent
        let mut sub_session = Session::new();
        let prompt = format!(
            "请为以下任务生成 Markdown 方案:\n目的: {}\n目标: {}\n意图: {}\n分解计划:\n{}",
            c.purpose, c.goal_summary, c.intent,
            c.decomposition_plan.iter().enumerate()
                .map(|(i, s)| format!("  {}. {}", i+1, s))
                .collect::<Vec<_>>().join("\n"),
        );
        sub_session.context_mut().push(ChatMessage::user(&prompt));
        let (text, usage) = self.agent.run_session(&mut sub_session).await?;

        // 3) 解析 Markdown, 验证完整性
        validate_plan_markdown(&text)?;

        // 4) 落盘
        let seq = self.db.next_seq(session.id())?;
        let path = plans_dir.join(format!("{}-{}.md", session.id(), seq));
        std::fs::write(&path, &text)?;

        Ok(path)
    }
}
```

```rust
impl QualityRunner {
    pub async fn check_subagent(
```

```rust
    ) -> Result<SessionSummary, AgentError> {
        let prompt = format!(
            "请对以下已完成任务生成简洁 Markdown 摘要(不超过 200 字)。\n目的: {}\n目标: {}\nAgent 链路: {}\nPlan 文档: {:?}\nWorkFlow: {}\n用量: input={}, output={}",
            c.purpose, c.goal_summary, c.agent_role.as_str(),
            result.plan_doc,
            result.workflows.iter().map(|w| format!("{}: {}", w.id, w.name)).collect::<Vec<_>>().join(", "),
            usage.input_tokens, usage.output_tokens,
        );

        let mut sub_session = Session::new();
        sub_session.context_mut().push(ChatMessage::user(&prompt));
        let (text, usage) = self.agent.run_session(&mut sub_session).await?;

        // 写入 session_memory
        let seq = self.db.next_seq(session.id())?;
        self.db.insert_memory(&SessionMemoryEntry {
            session_id: session.id().to_string(),
            seq,
            role: AgentRole::SessionContext,
            event_type: EventType::Summary,
            content: text.clone(),
            usage_input: usage.input_tokens,
            usage_output: usage.output_tokens,
        })?;

        Ok(SessionSummary { text, usage })
    }
}
```

## 11. Yolo 历史上下文注入

Yolo 处理前,SessionContextRunner::inject_history() 注入最近 N 条摘要:

```rust
const HISTORY_INJECT_MARKER_START: &str = "<<<LAEW:SESSION_HISTORY>>>";
const HISTORY_INJECT_MARKER_END: &str = "<<<LAEW:SESSION_HISTORY_END>>>";

pub fn build_history_message(entries: &[SessionMemoryEntry]) -> Option<ChatMessage> {
    if entries.is_empty() { return None; }
    let mut text = format!("{HISTORY_INJECT_MARKER_START}\n[SessionMemory 注入]\n");
    for e in entries {
        text.push_str(&format!("- [{}] seq={} role={} event={}\n  {}\n",
            e.created_at, e.seq, e.role.as_str(), e.event_type.as_str(),
            e.content.lines().next().unwrap_or("")));
    }
    text.push_str(HISTORY_INJECT_MARKER_END);
    Some(ChatMessage::user(text))
}

pub fn inject_history_once(session: &mut Session, entries: &[SessionMemoryEntry]) -> bool {
    if is_history_injected(session.context()) { return false; }
    if let Some(msg) = build_history_message(entries) {
        session.context_mut().insert(0, msg);
        true
    } else {
        false
    }
}
```

幂等探测:扫描 `MARKER_START` 字符串,避免重复注入。
